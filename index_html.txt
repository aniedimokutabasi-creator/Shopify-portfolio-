    <script>
        // Sample Product Data (Simulating a Shopify Storefront GraphQL Response)
        const products = [
            { id: 1, name: "Matte Clay Pomade", desc: "Strong hold, zero shine. Perfect for textured cuts.", price: 18.00, icon: "fa-pump-soap" },
            { id: 2, name: "Botanical Beard Oil", desc: "Nourishes skin underneath with organic jojoba oil.", price: 24.00, icon: "fa-bottle-droplet" },
            { id: 3, name: "Precision Edge Spray", desc: "Pre-shave cooling solution for sharp razor lines.", price: 15.00, icon: "fa-spray-can" },
            { id: 4, name: "Gold-Steel Trimmer Oil", desc: "Extends blade life and prevents thermal wear.", price: 12.00, icon: "fa-oil-well" }
        ];

        let cart = [];

        // Render Products
        function renderProducts() {
            const grid = document.getElementById('productGrid');
            grid.innerHTML = products.map(p => `
                <div class="product-card">
                    <div>
                        <div class="img-box"><i class="fa-solid ${p.icon}"></i></div>
                        <div class="product-title">${p.name}</div>
                        <div class="product-desc">${p.desc}</div>
                    </div>
                    <div class="product-bottom">
                        <div class="price">$${p.price.toFixed(2)}</div>
                        <button class="btn-add" onclick="addToCart(${p.id})">Add to Bag</button>
                    </div>
                </div>
            `).join('');
        }

        // Cart Drawer Logic
        function toggleCart() {
            document.getElementById('cartDrawer').classList.toggle('open');
            document.getElementById('cartOverlay').classList.toggle('open');
        }

        function addToCart(productId) {
            const product = products.find(p => p.id === productId);
            const existingItem = cart.find(item => item.id === productId);

            if (existingItem) {
                existingItem.qty += 1;
            } else {
                cart.push({ ...product, qty: 1 });
            }

            updateCartUI();
            toggleCart();
        }

        function updateCartUI() {
            const cartItemsContainer = document.getElementById('cartItems');
            const cartCount = document.getElementById('cartCount');
            const cartTotal = document.getElementById('cartTotal');

            const totalQty = cart.reduce((sum, item) => sum + item.qty, 0);
            const totalPrice = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);

            cartCount.textContent = totalQty;
            cartTotal.textContent = `$${totalPrice.toFixed(2)}`;

            if (cart.length === 0) {
                cartItemsContainer.innerHTML = '<p style="color: var(--text-muted); text-align: center; margin-top: 40px;">Your bag is empty.</p>';
            } else {
                cartItemsContainer.innerHTML = cart.map(item => `
                    <div class="cart-item">
                        <div>
                            <div style="font-weight:600; font-size:0.9rem;">${item.name}</div>
                            <div style="font-size:0.8rem; color:var(--text-muted);">$${item.price.toFixed(2)} x ${item.qty}</div>
                        </div>
                        <div style="font-weight:700; color:var(--accent);">$${(item.price * item.qty).toFixed(2)}</div>
                    </div>
                `).join('');
            }
        }

        // Initialize page
        renderProducts();
    </script>
